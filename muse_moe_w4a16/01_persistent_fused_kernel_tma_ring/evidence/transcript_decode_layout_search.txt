# Layouts of 8 E2M1 codes in a word such that 4 BF16x2 outputs (code bits at {6,7,8,15} + 16k) come from
# rotations of the word: 3 outputs with one rotation each, one output with two pieces. Cost model:
# LOP3 per piece; a shared temporary per distinct nonzero rotation: IMAD (left shift, FMA pipe) if no
# piece needs wrapped bits, else SHF (ALU).
import itertools
M = [6, 7, 8, 15, 22, 23, 24, 31]
FULL = (1 << 32) - 1
def srcmask(ps, r):
    m = 0
    for p in ps:
        m |= 1 << ((p - r) % 32)
    return m
def wraps(ps, r):
    return any(p - r < 0 for p in ps)
best = []
for r1, r2 in itertools.combinations(range(1, 32), 2):
    a = srcmask(M, 0); b = srcmask(M, r1); c = srcmask(M, r2)
    if a & b or a & c or b & c:
        continue
    rem = FULL & ~(a | b | c)
    for u, v in itertools.combinations(range(32), 2):
        # each p in M goes to u or v, sources exactly rem
        for bits in range(256):
            pu = [p for i, p in enumerate(M) if bits >> i & 1]
            pv = [p for i, p in enumerate(M) if not bits >> i & 1]
            if not pu or not pv:
                continue
            su, sv = srcmask(pu, u), srcmask(pv, v)
            if su & sv or (su | sv) != rem:
                continue
            # temporaries: rotations {r1, r2, u, v} minus 0, realized as IMAD if no wrap anywhere it is used
            uses = {}
            for r, ps in ((r1, M), (r2, M), (u, pu), (v, pv)):
                if r == 0:
                    continue
                uses.setdefault(r, []).append(ps)
            shf = sum(1 for r, L in uses.items() if any(wraps(ps, r) for ps in L))
            imad = len(uses) - shf
            lop = 5
            best.append((lop + shf, imad, (0, r1, r2), (u, v), tuple(pu), tuple(pv)))
best.sort()
print(len(best), "layouts")
seen = set()
for x in best[:15]:
