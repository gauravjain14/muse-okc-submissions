"""Per-kernel CUDA times of a submission, via torch.profiler over a few eager calls."""
import importlib.util, os, sys
import torch
HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, os.path.dirname(HERE))
import reference as R

spec = importlib.util.spec_from_file_location("sub", sys.argv[1])
sub = importlib.util.module_from_spec(spec); spec.loader.exec_module(sub)
weights = R.generate_weights(R.WEIGHTS_SEED)
state = sub.prepare(weights)
flush = torch.empty(512 * 1024 * 1024 // 4, device="cuda")
for bs in [int(b) for b in sys.argv[2].split(",")]:
    data = R.generate_input(bs, "ep8", 7)
    for _ in range(3):
        sub.kernel_fn(data, state)
    torch.cuda.synchronize()
    with torch.profiler.profile(activities=[torch.profiler.ProfilerActivity.CUDA]) as prof:
        for _ in range(5):
            flush.add_(1)
            sub.kernel_fn(data, state)
        torch.cuda.synchronize()
    tot = {}
    for ev in prof.events():
        if ev.device_type == torch.autograd.DeviceType.CUDA:
            tot.setdefault(ev.name, []).append(ev.device_time_total if hasattr(ev, "device_time_total") else ev.cuda_time_total)
    print(f"bs {bs}:")
    for k, v in sorted(tot.items(), key=lambda kv: -sum(kv[1])):
        print(f"   {sum(v)/len(v):9.1f} us x{len(v):3d}  {k[:90]}")
EOF

"""Registers and local-memory (spill) bytes of each compiled kernel."""
import importlib.util, sys
import torch
spec = importlib.util.spec_from_file_location("sub", sys.argv[1])
sub = importlib.util.module_from_spec(spec); spec.loader.exec_module(sub)
torch.zeros(1, device="cuda")
mod = sub._module()
d = mod.driver
names = ["moe_route", "moe_combine"] + [f"moe_gemm_{rw}_{nt}_{ks}_{ns}" for _, (rw, nt, ks, ns), _ in sub.CONFIGS]
for n in names:
    f = mod.func(n)
    regs = d.cuFuncGetAttribute(d.CUfunction_attribute.CU_FUNC_ATTRIBUTE_NUM_REGS, f)[1]
    local = d.cuFuncGetAttribute(d.CUfunction_attribute.CU_FUNC_ATTRIBUTE_LOCAL_SIZE_BYTES, f)[1]
    print(f"{n:24s} regs {regs:4d} local {local:6d} B")
